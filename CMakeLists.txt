# P-NET as ESP-IDF component

# Gere headers de opções e versão
set(PNET_MAX_AR 1)
set(PNET_MAX_API 1)
set(PNET_MAX_CR 2)
set(PNET_MAX_SLOTS 8)
set(PNET_MAX_SUBSLOTS 16)
set(PNET_MAX_DFP_IOCR 0)
set(PNET_MAX_PHYSICAL_PORTS 1)
set(PNET_MAX_LOG_BOOK_ENTRIES 32)
set(PNET_MAX_ALARMS 4)
set(PNET_MAX_ALARM_PAYLOAD_DATA_SIZE 512)
set(PNET_MAX_DIAG_ITEMS 32)
set(PNET_MAX_DIAG_MANUF_DATA_SIZE 64)
set(PNET_MAX_MC_CR 0)
set(PNET_MAX_AR_VENDOR_BLOCKS 0)
set(PNET_MAX_AR_VENDOR_BLOCK_DATA_LENGTH 0)
set(PNET_MAX_MAN_SPECIFIC_FAST_STARTUP_DATA_LENGTH 0)
set(PNET_MAX_SESSION_BUFFER_SIZE 2048)
set(PNET_MAX_DIRECTORYPATH_SIZE 128)
set(PNET_MAX_FILENAME_SIZE 64)
set(PNET_MAX_PORT_DESCRIPTION_SIZE 64)

set(PNET_OPTION_FAST_STARTUP 0)
set(PNET_OPTION_PARAMETER_SERVER 0)
set(PNET_OPTION_IR 0)
set(PNET_OPTION_SR 0)
set(PNET_OPTION_REDUNDANCY 0)
set(PNET_OPTION_AR_VENDOR_BLOCKS 0)
set(PNET_OPTION_RS 0)
set(PNET_OPTION_MC_CR 0)
set(PNET_OPTION_SRL 0)
set(PNET_OPTION_SNMP 0)
set(PNET_OPTION_DRIVER_ENABLE 0)
set(PNET_USE_ATOMICS 0)

set(LOG_LEVEL INFO)
set(PF_ETH_LOG ON)
set(PF_LLDP_LOG OFF)
set(PF_SNMP_LOG OFF)
set(PF_CPM_LOG OFF)
set(PF_PPM_LOG OFF)
set(PF_DCP_LOG ON)
set(PF_RPC_LOG ON)
set(PF_ALARM_LOG ON)
set(PF_AL_BUF_LOG OFF)
set(PF_PNAL_LOG ON)
set(PNET_LOG ON)

# Diretório de headers gerados
set(GEN_INC_DIR ${CMAKE_CURRENT_BINARY_DIR}/include)
file(MAKE_DIRECTORY ${GEN_INC_DIR})

configure_file(${CMAKE_CURRENT_LIST_DIR}/pnet_options.h.in ${GEN_INC_DIR}/pnet_options.h @ONLY)
configure_file(${CMAKE_CURRENT_LIST_DIR}/pnet_version.h.in ${GEN_INC_DIR}/pnet_version.h @ONLY)

# Fontes
file(GLOB_RECURSE PNET_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/src/*.c
)

# Excluir drivers específicos do Linux que não funcionam no ESP32
list(FILTER PNET_SOURCES EXCLUDE REGEX ".*lan9662.*")
list(FILTER PNET_SOURCES EXCLUDE REGEX ".*rte_uio.*")

# Excluir SNMP se estiver desabilitado
if(NOT PNET_OPTION_SNMP)
    list(FILTER PNET_SOURCES EXCLUDE REGEX ".*snmp.*")
endif()

idf_component_register(
    SRCS ${PNET_SOURCES}
    INCLUDE_DIRS 
        ${CMAKE_CURRENT_LIST_DIR}/include 
        ${GEN_INC_DIR} 
        ${CMAKE_CURRENT_LIST_DIR}/src 
        ${CMAKE_CURRENT_LIST_DIR}/src/common 
        ${CMAKE_CURRENT_LIST_DIR}/src/device
)
